// 定位微信主窗口（关键！）
bool ATSPIEngine::init(const std::string& app_name) {
    desktop_ = atspi_get_desktop(0);
    if (!desktop_) return false;
    
    // 遍历所有应用，查找微信
    for (int i = 0; i < atspi_accessible_get_child_count(desktop_, nullptr); ++i) {
        auto app = atspi_accessible_get_child_at_index(desktop_, i, nullptr);
        if (!app) continue;
        
        char* name = atspi_accessible_get_name(app, nullptr);
        if (name && (strstr(name, "WeChat") || strstr(name, "微信"))) {
            wechat_app_ = app;
            
            // 定位主窗口：ClassName="WeChatMainWndForPC"
            auto window = find_control({{"role", "frame"}, {"name", "微信"}});
            if (window) {
                main_window_ = window;
                g_free(name);
                return true;
            }
        }
        g_free(name);
    }
    return false;
}

// 递归搜索控件（微信4.1.0联系人列表结构）
std::vector<ControlInfo> ATSPIEngine::get_contact_list() {
    std::vector<ControlInfo> contacts;
    
    // 1. 点击"通讯录"标签（role="page tab", name="通讯录"）
    auto contact_tab = find_control({{"role", "page tab"}, {"name", "通讯录"}});
    if (contact_tab) {
        click_control(contact_tab);
        g_usleep(800000); // 800ms
    }
    
    // 2. 查找联系人列表容器（role="list"）
    auto list_container = find_control({{"role", "list"}}, 5, main_window_);
    if (!list_container) return contacts;
    
    // 3. 遍历子项（每个联系人是"list item"）
    int child_count = atspi_accessible_get_child_count(list_container, nullptr);
    for (int i = 0; i < std::min(child_count, 200); ++i) { // 防卡死
        auto item = atspi_accessible_get_child_at_index(list_container, i, nullptr);
        if (!item) continue;
        
        char* role = atspi_accessible_get_role_name(item, nullptr);
        if (role && strcmp(role, "list item") == 0) {
            ControlInfo info = get_control_info(item);
            
            // 提取微信ID（关键！微信Qt将ID存于description）
            info.description = _extract_wechat_id(item);
            
            // 获取昵称：第一个text子控件
            auto texts = find_all_children(item, "text");
            if (!texts.empty()) {
                info.name = get_control_info(texts[0]).name;
            }
            
            contacts.push_back(info);
        }
        g_free(role);
    }
    return contacts;
}

// 微信ID提取（微信4.1.0实测）
std::string ATSPIEngine::_extract_wechat_id(AtspiAccessible* item) {
    char* desc = atspi_accessible_get_description(item, nullptr);
    if (desc && strstr(desc, "wxid_")) {
        // 格式示例: "contact_wxid_wxid_xxx"
        std::string s(desc);
        size_t pos = s.find("wxid_");
        if (pos != std::string::npos) {
            size_t end = s.find_first_of(" \t\n", pos);
            g_free(desc);
            return s.substr(pos, end != std::string::npos ? end - pos : std::string::npos);
        }
    }
    g_free(desc);
    return "unknown_" + std::to_string((long)item);
}

from abc import ABC, abstractmethod
import subprocess
import re

class WeChatAdapter(ABC):
    @abstractmethod
    def get_contact_list(self): pass
    @abstractmethod
    def send_message_by_id(self, wechat_id, content): pass
    @abstractmethod
    def get_chat_history_by_id(self, wechat_id, max_count): pass

class WeChat410Adapter(WeChatAdapter):
    """微信4.1.0专用适配器（AT-SPI主引擎）"""
    def __init__(self):
        from wechat_atspi import ATSPIEngine  # C++绑定模块
        self.engine = ATSPIEngine()
        if not self.engine.init():
            raise RuntimeError("AT-SPI初始化失败，请确认：1.微信已启动 2.辅助功能已开启")
    
    def get_contact_list(self):
        return self.engine.get_contact_list()  # 直接调用C++引擎
    
    def send_message_by_id(self, wechat_id, content):
        # 1. 获取联系人列表
        contacts = self.get_contact_list()
        # 2. 查找匹配wechat_id的联系人
        target = next((c for c in contacts if wechat_id in c.description), None)
        if not target:
            return False
        # 3. 发送消息（C++引擎内部处理点击+输入+发送）
        return self.engine.send_message_to_contact(target, content)
    
    def get_chat_history_by_id(self, wechat_id, max_count):
        # 先切换到该联系人聊天窗口
        if self.send_message_by_id(wechat_id, ""):  # 空消息仅用于切换窗口
            return self.engine.get_chat_history(max_count)
        return []

# humanization.py
import time, random, math

def human_delay(base=1.0, variance=0.5):
    """符合人类操作分布的延迟"""
    # 使用正态分布（均值base，标准差variance）
    delay = max(0.3, random.gauss(base, variance))
    time.sleep(delay)
    return delay

def human_typing(text, engine):
    """模拟人类打字（含错误修正）"""
    for i, char in enumerate(text):
        # 5%概率模拟打字错误（仅非标点字符）
        if random.random() < 0.05 and char.isalnum():
            wrong_char = chr(random.randint(97, 122)) # 随机字母
            engine.press_key(wrong_char)
            human_delay(0.1, 0.05)
            engine.press_key("backspace") # 删除
            human_delay(0.2, 0.1)
        
        engine.press_key(char)
        # 打字间隔：字母快，标点慢
        delay = 0.08 if char.isalnum() else 0.3
        human_delay(delay, delay*0.3)

# safety_limits.py
OPERATION_LIMITS = {
    "messages_per_hour": 15,    # 每小时消息上限
    "contacts_per_session": 30, # 单次会话联系人上限
    "min_interval_sec": 45,     # 消息最小间隔（秒）
    "max_session_min": 90       # 单次会话最大时长（分钟）
}

class SafetyMonitor:
    def __init__(self):
        self.operation_log = [] # (timestamp, operation_type)
    
    def check_allow(self, op_type):
        now = time.time()
        # 清理1小时前的操作
        self.operation_log = [log for log in self.operation_log if now - log[0] < 3600]
        
        # 检查小时限额
        if op_type == "message" and len([l for l in self.operation_log if l[1]=="message"]) >= OPERATION_LIMITS["messages_per_hour"]:
            raise SafetyException("达到每小时消息上限")
        
        # 检查最小间隔
        if self.operation_log and now - self.operation_log[-1][0] < OPERATION_LIMITS["min_interval_sec"]:
            wait = OPERATION_LIMITS["min_interval_sec"] - (now - self.operation_log[-1][0])
            time.sleep(wait)
        
        self.operation_log.append((now, op_type))
